for frontend setup:-
cd frontend
npm create vite@latest .  # if not done already
npm install


for run:- (Note please check you are in frontend folder in terminal)
npm run dev

for backend setup:-
cd backend
python -m venv venv
venv\Scripts\activate    # Windows
pip install fastapi uvicorn python-multipart openai

for run:- (Note please check you are in backend folder in terminal)
uvicorn src.api.app:app --reload
